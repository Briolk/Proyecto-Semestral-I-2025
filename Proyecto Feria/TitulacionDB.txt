CREATE DATABASE gestion_titulacion;
USE gestion_titulacion;

-- Tabla para Modalidades de Graduación
CREATE TABLE modalidades_graduacion (
    id_modalidad INT AUTO_INCREMENT PRIMARY KEY,
    nombre_modalidad VARCHAR(50) NOT NULL,
    descripcion TEXT
);

-- Tabla para Registro de Estudiantes
CREATE TABLE estudiantes (
    ci VARCHAR(20) PRIMARY KEY,
    ru VARCHAR(20) NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    apellidos VARCHAR(50) NOT NULL,
    correo_electronico VARCHAR(50),
    estado VARCHAR(20) NOT NULL,
    id_modalidad INT NOT NULL, 
    FOREIGN KEY (id_modalidad) REFERENCES modalidades_graduacion(id_modalidad)
);

-- Tabla para Asignación de Docente Tutor
CREATE TABLE tutores (
    id_tutor INT AUTO_INCREMENT PRIMARY KEY,
    nombre_tutor VARCHAR(50) NOT NULL,
    especialidad VARCHAR(50)
);

CREATE TABLE asignacion_tutor (
    id_asignacion INT AUTO_INCREMENT PRIMARY KEY,
    ci_estudiante VARCHAR(20),
    id_tutor INT,
    fecha_asignacion DATE,
    FOREIGN KEY (ci_estudiante) REFERENCES estudiantes(ci),
    FOREIGN KEY (id_tutor) REFERENCES tutores(id_tutor)
);

-- Tabla para Fases de Evaluación
CREATE TABLE fases_evaluacion (
    id_fase INT AUTO_INCREMENT PRIMARY KEY,
    nombre_fase VARCHAR(50) NOT NULL,
    descripcion TEXT,
    orden INT NOT NULL
);

-- Tabla para Seguimiento de Avance de Evaluación
CREATE TABLE avance_evaluacion (
    id_avance INT AUTO_INCREMENT PRIMARY KEY,
    ci_estudiante VARCHAR(20),
    id_fase INT,
    fecha_aprobacion DATE,
    estado_fase VARCHAR(20) DEFAULT 'Pendiente',
    FOREIGN KEY (ci_estudiante) REFERENCES estudiantes(ci),
    FOREIGN KEY (id_fase) REFERENCES fases_evaluacion(id_fase)
);

-- Tabla para Registro de Pagos por Titulación, relacionada con Fases de Evaluación
CREATE TABLE pagos_titulacion (
    id_pago INT AUTO_INCREMENT PRIMARY KEY,
    ci_estudiante VARCHAR(20),
    id_fase INT,
    monto DECIMAL(10, 2) NOT NULL,
    fecha_pago DATE,
    estado_pago VARCHAR(20),
    FOREIGN KEY (ci_estudiante) REFERENCES estudiantes(ci),
    FOREIGN KEY (id_fase) REFERENCES fases_evaluacion(id_fase)
);
